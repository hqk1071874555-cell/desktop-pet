### 2.2 键盘/鼠标监听 (src-tauri/src/input.rs)

```rust
use rdev::{listen, Event, EventType};
use tauri::Emitter;
use std::sync::Mutex;
use std::thread;

pub struct InputState {
    listening: Mutex<bool>,
}

impl InputState {
    pub fn new() -> Self {
        Self { listening: Mutex::new(false) }
    }

    /// 启动全局键盘/鼠标监听（在独立线程中运行）
    pub fn start(app_handle: tauri::AppHandle) {
        thread::spawn(move || {
            if let Err(error) = listen(move |event: Event| {
                let app = app_handle.clone();
                match event.event_type {
                    EventType::KeyPress(key) => {
                        let key_name = format!("{:?}", key);
                        let _ = app.emit("device-changed", serde_json::json!({
                            "kind": "KeyboardPress",
                            "value": key_name
                        }));
                    }
                    EventType::KeyRelease(key) => {
                        let key_name = format!("{:?}", key);
                        let _ = app.emit("device-changed", serde_json::json!({
                            "kind": "KeyboardRelease",
                            "value": key_name
                        }));
                    }
                    EventType::ButtonPress(button) => {
                        let _ = app.emit("device-changed", serde_json::json!({
                            "kind": "MousePress",
                            "value": format!("{:?}", button)
                        }));
                    }
                    EventType::ButtonRelease(button) => {
                        let _ = app.emit("device-changed", serde_json::json!({
                            "kind": "MouseRelease",
                            "value": format!("{:?}", button)
                        }));
                    }
                    EventType::MouseMove { x, y } => {
                        let _ = app.emit("device-changed", serde_json::json!({
                            "kind": "MouseMove",
                            "x": x,
                            "y": y
                        }));
                    }
                    _ => {}
                }
            }) {
                eprintln!("rdev listen error: {:?}", error);
            }
        });
    }
}
```

### 2.3 手柄监听 (src-tauri/src/gamepad.rs)

```rust
use gilrs::{Gilrs, Event, EventType};
use tauri::Emitter;
use std::thread;
use std::time::Duration;

/// 启动手柄轮询（在独立线程中运行，每 16ms 轮询一次 ≈ 60FPS）
pub fn start_gamepad(app_handle: tauri::AppHandle) {
    thread::spawn(move || {
        let mut gilrs = match Gilrs::new() {
            Ok(g) => g,
            Err(e) => {
                eprintln!("Failed to init gilrs: {:?}", e);
                return;
            }
        };

        loop {
            // 检查是否有新的手柄事件
            while let Some(Event { event, .. }) = gilrs.next_event() {
                match event {
                    EventType::AxisChanged(axis, value, ..) => {
                        let _ = app_handle.emit("gamepad-changed", serde_json::json!({
                            "type": "Axis",
                            "axis": format!("{:?}", axis),
                            "value": value  // -1.0 ~ 1.0
                        }));
                    }
                    EventType::ButtonPressed(button, ..) => {
                        let _ = app_handle.emit("gamepad-changed", serde_json::json!({
                            "type": "ButtonPress",
                            "button": format!("{:?}", button)
                        }));
                    }
                    EventType::ButtonReleased(button, ..) => {
                        let _ = app_handle.emit("gamepad-changed", serde_json::json!({
                            "type": "ButtonRelease",
                            "button": format!("{:?}", button)
                        }));
                    }
                    _ => {}
                }
            }
            thread::sleep(Duration::from_millis(16));
        }
    });
}
```

export interface KeyboardEvent {
  kind: 'KeyboardPress' | 'KeyboardRelease';
  value: string; // 键名，如 "KeyA", "Space", "Enter"
}

// 鼠标事件
export interface MouseEvent {
  kind: 'MousePress' | 'MouseRelease' | 'MouseMove';
  value?: string;
  x?: number;
  y?: number;
}

// 手柄轴事件
export interface GamepadAxisEvent {
  type: 'Axis';
  axis: string;   // "LeftStickX", "LeftStickY", "RightStickX", "RightStickY"
  value: number;  // -1.0 ~ 1.0
}

// 手柄按钮事件
export interface GamepadButtonEvent {
  type: 'ButtonPress' | 'ButtonRelease';
  button: string; // "South"(A), "East"(B), "West"(X), "North"(Y), "LeftTrigger", etc.
}

export type DeviceEvent = KeyboardEvent | MouseEvent;
export type GamepadEvent = GamepadAxisEvent | GamepadButtonEvent;

### 3.2 useDevice Hook (src/hooks/useDevice.ts)

```typescript
import { useEffect, useRef, useCallback } from 'react';
import { listen } from '@tauri-apps/api/event';
import type { DeviceEvent } from '../types/input';

interface UseDeviceOptions {
  onKeyPress?: (key: string) => void;
  onKeyRelease?: (key: string) => void;
  onMouseMove?: (x: number, y: number) => void;
  onMousePress?: (button: string) => void;
  onMouseRelease?: (button: string) => void;
  autoReleaseDelay?: number; // Windows 自动释放延迟 (ms)
}

// 需要自动释放的特殊键（CapsLock 等切换键）
const TOGGLE_KEYS = new Set(['CapsLock', 'NumLock', 'ScrollLock']);

// 按键标准化映射表
const KEY_NORMALIZE_MAP: Record<string, string> = {
  'KeyA': 'A', 'KeyB': 'B', 'KeyC': 'C', 'KeyD': 'D', 'KeyE': 'E',
  'KeyF': 'F', 'KeyG': 'G', 'KeyH': 'H', 'KeyI': 'I', 'KeyJ': 'J',
  'KeyK': 'K', 'KeyL': 'L', 'KeyM': 'M', 'KeyN': 'N', 'KeyO': 'O',
  'KeyP': 'P', 'KeyQ': 'Q', 'KeyR': 'R', 'KeyS': 'S', 'KeyT': 'T',
  'KeyU': 'U', 'KeyV': 'V', 'KeyW': 'W', 'KeyX': 'X', 'KeyY': 'Y',
  'KeyZ': 'Z',
  'Digit0': '0', 'Digit1': '1', 'Digit2': '2', 'Digit3': '3', 'Digit4': '4',
  'Digit5': '5', 'Digit6': '6', 'Digit7': '7', 'Digit8': '8', 'Digit9': '9',
  'Space': 'Space', 'Enter': 'Enter', 'Backspace': 'Backspace',
  'Tab': 'Tab', 'Escape': 'Escape',
  'ShiftLeft': 'Shift', 'ShiftRight': 'Shift',
  'ControlLeft': 'Ctrl', 'ControlRight': 'Ctrl',
  'AltLeft': 'Alt', 'AltRight': 'Alt',
  'MetaLeft': 'Meta', 'MetaRight': 'Meta',
  'ArrowUp': 'Up', 'ArrowDown': 'Down',
  'ArrowLeft': 'Left', 'ArrowRight': 'Right',
  'F1': 'F1', 'F2': 'F2', 'F3': 'F3', 'F4': 'F4',
  'F5': 'F5', 'F6': 'F6', 'F7': 'F7', 'F8': 'F8',
  'F9': 'F9', 'F10': 'F10', 'F11': 'F11', 'F12': 'F12',
};

function normalizeKey(rawKey: string): string {
  return KEY_NORMALIZE_MAP[rawKey] || rawKey;
}

export function useDevice(options: UseDeviceOptions = {}) {
  const {
    onKeyPress, onKeyRelease,
    onMouseMove, onMousePress, onMouseRelease,
    autoReleaseDelay = 200,
  } = options;

  const releaseTimers = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map());
  const isWindows = useRef(navigator.platform.includes('Win'));

  const handleAutoRelease = useCallback((key: string, delay: number) => {
    // 清除已有的定时器
    const existing = releaseTimers.current.get(key);
    if (existing) clearTimeout(existing);

    // 设置自动释放定时器
    const timer = setTimeout(() => {
      onKeyRelease?.(key);
      releaseTimers.current.delete(key);
    }, delay);
    releaseTimers.current.set(key, timer);
  }, [onKeyRelease]);

  useEffect(() => {
    const unlisten = listen<DeviceEvent>('device-changed', (event) => {
      const { kind, value } = event.payload;

      // === 键盘事件 ===
      if (kind === 'KeyboardPress') {
        const key = normalizeKey(value!);

        if (TOGGLE_KEYS.has(key)) {
          // 切换键：100ms 自动释放
          onKeyPress?.(key);
          handleAutoRelease(key, 100);
        } else if (isWindows.current) {
          // Windows：所有按键自动释放
          onKeyPress?.(key);
          handleAutoRelease(key, autoReleaseDelay);
        } else {
          // macOS/Linux：直接按下
          onKeyPress?.(key);
        }
      }

      if (kind === 'KeyboardRelease') {
        const key = normalizeKey(value!);
        if (!isWindows.current && !TOGGLE_KEYS.has(key)) {
          onKeyRelease?.(key);
        }
      }

      // === 鼠标事件 ===
      if (kind === 'MousePress' && value) {
        onMousePress?.(value);
      }
      if (kind === 'MouseRelease' && value) {
        onMouseRelease?.(value);
      }
      if (kind === 'MouseMove') {
        onMouseMove?.(event.payload.x!, event.payload.y!);
      }
    });

    return () => {
      unlisten.then(fn => fn());
      // 清理所有定时器
      releaseTimers.current.forEach(timer => clearTimeout(timer));
      releaseTimers.current.clear();
    };
  }, []); // 只在挂载时注册一次
}
```

### 3.3 useGamepad Hook (src/hooks/useGamepad.ts)

```typescript
import { useEffect, useCallback } from 'react';
import { listen } from '@tauri-apps/api/event';
import type { GamepadEvent } from '../types/input';

interface UseGamepadOptions {
  onAxisChange?: (axis: string, value: number) => void;
  onButtonPress?: (button: string) => void;
  onButtonRelease?: (button: string) => void;
  deadZone?: number; // 摇杆死区，默认 0.15
}

// 手柄按钮 → 键盘等价映射（让手柄也能触发键盘动画）
const BUTTON_TO_KEY_MAP: Record<string, string> = {
  'South': 'A',      // A 按钮 → 显示 A
  'East': 'B',       // B 按钮 → 显示 B
  'West': 'X',       // X 按钮 → 显示 X
  'North': 'Y',      // Y 按钮 → 显示 Y
  'DPadUp': 'Up',
  'DPadDown': 'Down',
  'DPadLeft': 'Left',
  'DPadRight': 'Right',
  'LeftTrigger': 'LT',
  'RightTrigger': 'RT',
  'LeftThumb': 'L3',
  'RightThumb': 'R3',
  'Start': 'Start',
  'Select': 'Select',
};

export function useGamepad(options: UseGamepadOptions = {}) {
  const { onAxisChange, onButtonPress, onButtonRelease, deadZone = 0.15 } = options;

  useEffect(() => {
    const unlisten = listen<GamepadEvent>('gamepad-changed', (event) => {
      const { type } = event.payload;

      if (type === 'Axis') {
        const { axis, value } = event.payload as GamepadAxisEvent;
        // 死区过滤：微小抖动忽略
        if (Math.abs(value) < deadZone) {
          onAxisChange?.(axis, 0);
        } else {
          onAxisChange?.(axis, value);
        }
      }

      if (type === 'ButtonPress') {
        const { button } = event.payload as GamepadButtonEvent;
        onButtonPress?.(button);
        // 手柄按钮映射为按键（复用按键动画系统）
        const mappedKey = BUTTON_TO_KEY_MAP[button];
        if (mappedKey) {
          onButtonPress?.(`Gamepad_${mappedKey}`);
        }
      }

      if (type === 'ButtonRelease') {
        const { button } = event.payload as GamepadButtonEvent;
        onButtonRelease?.(button);
        const mappedKey = BUTTON_TO_KEY_MAP[button];
        if (mappedKey) {
          onButtonRelease?.(`Gamepad_${mappedKey}`);
        }
      }
    });

    return () => { unlisten.then(fn => fn()); };
  }, [deadZone]);
}
```

### 3.4 集成 Live2D 动画 (src/hooks/usePetInput.ts)

```typescript
import { useRef, useState } from 'react';
import { useDevice } from './useDevice';
import { useGamepad } from './useGamepad';
import { useLive2D } from './useLive2D'; // 假设已有 Live2D Hook

interface PetInputOptions {
  modelRef: React.RefObject<Live2DModel | null>;
}

export function usePetInput({ modelRef }: PetInputOptions) {
  const pressedKeys = useRef<Set<string>>(new Set());
  const [displayKeys, setDisplayKeys] = useState<string[]>([]);

  // === 键盘/鼠标处理 ===
  useDevice({
    onKeyPress: (key) => {
      pressedKeys.current.add(key);
      setDisplayKeys(Array.from(pressedKeys.current));
      // 触发 Live2D 手部动画
      modelRef.current?.motion(`key_${key.toLowerCase()}`);
    },
    onKeyRelease: (key) => {
      pressedKeys.current.delete(key);
      setDisplayKeys(Array.from(pressedKeys.current));
    },
    onMouseMove: (x, y) => {
      // 驱动 Live2D 眼球追踪
      modelRef.current?.setParameter('ParamEyeBallX', x);
      modelRef.current?.setParameter('ParamEyeBallY', y);
    },
    onMousePress: (button) => {
      // 鼠标点击 → 触发动感
      modelRef.current?.motion('click');
    },
  });

  // === 手柄处理 ===
  useGamepad({
    deadZone: 0.15,
    onAxisChange: (axis, value) => {
      // 摇杆 → 身体倾斜
      if (axis === 'LeftStickX') {
        modelRef.current?.setParameter('ParamBodyAngleX', value * 30);
      }
      if (axis === 'LeftStickY') {
        modelRef.current?.setParameter('ParamBodyAngleY', value * 10);
      }
    },
    onButtonPress: (button) => {
      // 手柄按钮 → 显示按键 + 触发动作
      pressedKeys.current.add(`🎮${button}`);
      setDisplayKeys(Array.from(pressedKeys.current));
    },
    onButtonRelease: (button) => {
      pressedKeys.current.delete(`🎮${button}`);
      setDisplayKeys(Array.from(pressedKeys.current));
    },
  });

  return { displayKeys, pressedKeys };
}
```
