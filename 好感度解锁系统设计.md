# 好感度解锁系统 — 设计参考

> 键盘/手柄输入驱动，每1000次输入随机解锁一个表情差分

---

## 一、系统规则

```
输入次数 → 好感等级 → 解锁表情

每 1000 次输入 = 1 级好感
每升 1 级 = 随机解锁 1 个未拥有的表情
共 7 种表情：全部解锁需要 7000 次输入

🔒 特殊规则：爱心眼 (love) 固定最后解锁
   前 6 级：从其余 5 个表情中随机
   第 7 级：解锁爱心眼

键盘按下 = 1 次
手柄按钮  = 1 次
鼠标点击  = 0.5 次（防刷）
摇杆动作  = 不计入（避免误触）
```

---

## 二、7 种表情差分

| ID | 表情名 | 图标 | 初始 |
|----|--------|------|------|
| `normal` | 普通微笑 | 😊 | ✅ 默认拥有 |
| `love` | 爱心眼 | 😍 | 🔒 最后解锁 |
| `angry` | 愤怒冒火 | 😠 | 🔒 待解锁 |
| `surprised` | 惊讶 | 😲 | 🔒 待解锁 |
| `sad` | 悲伤 | 😢 | 🔒 待解锁 |
| `annoyed` | 翻白眼 | 🙄 | 🔒 待解锁 |
| `shocked` | 惊恐灰白 | 😱 | 🔒 待解锁 |

## 三、数据模型

### Rust 端

```rust
// src-tauri/src/affection.rs
use serde::{Deserialize, Serialize};
use rand::seq::SliceRandom;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AffectionState {
    pub total_inputs: u64,
    pub level: u32,
    pub unlocked_expressions: Vec<String>,
    pub inputs_to_next: u32,
    expression_pool: Vec<String>,
}

impl Default for AffectionState {
    fn default() -> Self {
        Self {
            total_inputs: 0,
            level: 0,
            unlocked_expressions: vec!["normal".to_string()],
            inputs_to_next: 1000,
            expression_pool: vec![
                "love", "angry", "surprised",
                "sad", "annoyed", "shocked",
            ].into_iter().map(String::from).collect(),
        }
    }
}

impl AffectionState {
    /// 记录输入。返回 Some(表情名) 表示解锁了新表情
    pub fn add_input(&mut self, count: f32) -> Option<String> {
        self.total_inputs = (self.total_inputs as f32 + count) as u64;
        let new_level = (self.total_inputs / 1000) as u32;

        if new_level > self.level {
            self.level = new_level;

            // 前 6 级：从非 love 的待解锁表情中随机选
            // 第 7 级（只剩 love 时）：解锁 love
            let pool: Vec<&String> = if self.unlocked_expressions.len() >= 6 {
                // 只剩 love 了
                vec![&self.expression_pool.iter()
                    .find(|e| **e == "love")
                    .unwrap()]
            } else {
                // 从排除 love 的池子里随机选
                self.expression_pool
                    .iter()
                    .filter(|e| **e != "love" && !self.unlocked_expressions.contains(e))
                    .collect()
            };

            let unlocked = pool.choose(&mut rand::thread_rng())
                .map(|s| s.to_string());

            if let Some(ref expr) = unlocked {
                self.unlocked_expressions.push(expr.clone());
            }

            self.inputs_to_next = 1000 - (self.total_inputs % 1000) as u32;
            return unlocked;
        }

        self.inputs_to_next = 1000 - (self.total_inputs % 1000) as u32;
        None
    }
}
```

## 四、核心 Hook

```typescript
// src/hooks/useAffection.ts
import { useState, useEffect } from 'react';
import { listen } from '@tauri-apps/api/event';

const STORAGE_KEY = 'desktop-pet-affection';

function loadState(): AffectionState {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : {
      totalInputs: 0, level: 0,
      unlockedExpressions: ['normal'], inputsToNext: 1000,
    };
  } catch { /* fallback to default */ }
  return { totalInputs: 0, level: 0, unlockedExpressions: ['normal'], inputsToNext: 1000 };
}

export function useAffection() {
  const [state, setState] = useState<AffectionState>(loadState);
  const [lastUnlocked, setLastUnlocked] = useState<string | null>(null);
  const [showUnlock, setShowUnlock] = useState(false);

  useEffect(() => {
    const setup = listen<{ count: number }>('affection-input', (event) => {
      setState(prev => {
        const newTotal = prev.totalInputs + event.payload.count;
        const newLevel = Math.floor(newTotal / 1000);
        const updated = {
          totalInputs: newTotal,
          level: newLevel,
          unlockedExpressions: [...prev.unlockedExpressions],
          inputsToNext: 1000 - (newTotal % 1000),
        };

        if (newLevel > prev.level) {
          // 前 6 级：从非 love 的表情中随机
          // 第 7 级（只剩 love）：解锁 love
          const pool = updated.unlockedExpressions.length >= 6
            ? ['love']  // 只剩下爱心眼了
            : ['angry','surprised','sad','annoyed','shocked']
                .filter(e => !updated.unlockedExpressions.includes(e));
          if (pool.length > 0) {
            const unlocked = pool[Math.floor(Math.random() * pool.length)];
            updated.unlockedExpressions.push(unlocked);
            setLastUnlocked(unlocked);
            setShowUnlock(true);
            setTimeout(() => setShowUnlock(false), 3000);
          }
        }

        localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
        return updated;
      });
    });
    return () => { setup.then(f => f?.()); };
  }, []);

  return {
    ...state,
    lastUnlocked, showUnlock,
    progressPercent: Math.round(((state.totalInputs % 1000) / 1000) * 100),
  };
}
```

## 五、进度条组件

```tsx
// src/components/AffectionBar.tsx
import { useAffection } from '../hooks/useAffection';
import { EXPRESSION_NAMES } from '../types/affection';

const EMOJI_MAP: Record<string, string> = {
  normal: '😊', love: '😍', angry: '😠', surprised: '😲',
  sad: '😢', annoyed: '🙄', shocked: '😱',
};

export function AffectionBar() {
  const { level, totalInputs, inputsToNext, progressPercent,
          unlockedExpressions, showUnlock, lastUnlocked } = useAffection();

  return (
    <div className="affection-bar">
      {/* 等级 */}
      <div className="affection-header">
        <span>❤️ Lv.{level}</span>
        <span>{totalInputs} 次</span>
      </div>

      {/* 进度条 */}
      <div className="affection-progress">
        <div className="affection-progress-fill"
             style={{ width: `${progressPercent}%` }} />
      </div>
      <div className="affection-next">距下一级还需 {inputsToNext} 次</div>

      {/* 已解锁表情 */}
      <div className="affection-expressions">
        {unlockedExpressions.map(id => (
          <span key={id} title={EXPRESSION_NAMES[id]}>{EMOJI_MAP[id]}</span>
        ))}
        {Array.from({ length: 7 - unlockedExpressions.length }).map((_, i) => (
          <span key={i} className="locked">🔒</span>
        ))}
      </div>

      {/* 解锁动画 */}
      {showUnlock && lastUnlocked && (
        <div className="affection-unlock-toast">
          🎉 解锁「{EXPRESSION_NAMES[lastUnlocked]}」{EMOJI_MAP[lastUnlocked]}
        </div>
      )}
    </div>
  );
}
```

## 七、事件流总览

```
键盘按下 / 手柄按钮
      ↓
 rdev / gilrs (Rust)
      ↓
 emit("affection-input", { count: 1.0 })
      ↓
 useAffection Hook (React)
      ↓
 totalInputs += count
      ↓
 newLevel > currentLevel ?
  ├─ NO:  更新进度条
  └─ YES: 随机选未解锁表情
           ├─ localStorage 持久化
           ├─ 3秒解锁动画
           ├─ Live2D 庆祝动作
           └─ 切换为新表情
```
